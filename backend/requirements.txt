# ============================================================================
# FILE:       backend/requirements.txt
# PURPOSE:    Pinned list of every Python package the TaskFlow backend needs.
#             `pip install -r requirements.txt` reproduces the exact
#             environment on any machine (reproducibility).
# HOW IT WORKS: One package per line as `name==exact.version`. Sections group
#             packages by the job they do. Explanations of each package are in
#             docs/CONFIG_EXPLAINED.md section 1.
# USED BY / DEPENDS ON: installed into backend/.venv before running anything.
# KEY CONCEPTS: dependency pinning, dev vs runtime dependencies.
# ============================================================================

# --- Web framework ---------------------------------------------------------
# fastapi gives async HTTP routes + automatic OpenAPI docs from type hints
fastapi==0.115.6
# uvicorn is the ASGI server that actually runs the FastAPI app (dev: --reload)
uvicorn[standard]==0.32.1
# python-multipart lets FastAPI parse multipart/form-data (CSV/JSON file upload)
python-multipart==0.0.19

# --- Database layer --------------------------------------------------------
# sqlalchemy 2.0 is the ORM: Python classes mapped to tables
SQLAlchemy==2.0.36
# asyncpg is the non-blocking PostgreSQL driver used by the async engine
asyncpg==0.30.0
# alembic is the migration tool: versioned scripts for schema changes
alembic==1.14.0

# --- Configuration & validation --------------------------------------------
# pydantic v2 validates request/response data and powers the OpenAPI schema
pydantic==2.10.3
# pydantic-settings loads .env files into typed Python settings objects
pydantic-settings==2.6.1
# email-validator makes Pydantic's EmailStr type work (unique user emails)
email-validator==2.2.0

# --- Runtime helpers --------------------------------------------------------
# python-dotenv is used internally by pydantic-settings to read .env
python-dotenv==1.0.1

# --- Testing ----------------------------------------------------------------
# pytest is the test runner (discovers files named test_*.py)
pytest==8.3.4
# pytest-asyncio lets pytest execute `async def` test functions
pytest-asyncio==0.24.0
# httpx gives AsyncClient: calls the FastAPI app in-process, no live server
httpx==0.28.1
# anyio is required by FastAPI/httpx for async compatibility layers
anyio==4.7.0
